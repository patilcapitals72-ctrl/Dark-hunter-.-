const express = require('express'); const mongoose = require('mongoose'); const cors = require('cors'); const app = express(); app.use(express.json()); app.use(cors()); // 1. Database Connection (MongoDB) // Replace 'YOUR_MONGODB_URI' with your real database link mongoose.connect('mongodb://localhost:27017/tradefast_pro') .then(() => console.log("Connected to Secure Database")) .catch(err => console.error("Database Connection Error:", err)); // 2. User Data Model const userSchema = new mongoose.Schema({ username: String, upiId: String, balance: { type: Number, default: 0 }, tapCoins: { type: Number, default: 0 }, isPremium: { type: Boolean, default: false }, dailyTaps: { type: Number, default: 1000 }, lastTapDate: { type: Date, default: Date.now } }); const User = mongoose.model('User', userSchema); // 3. API - Handling User Taps (Logic) app.post('/api/tap', async (req, res) => { const { userId } = req.body; const user = await User.findById(userId); if (!user.isPremium) { return res.status(403).json({ message: "Access Denied: Payment Required" }); } if (user.dailyTaps > 0) { user.tapCoins += 0.000001; user.dailyTaps -= 1; await user.save(); res.json({ coins: user.tapCoins, remaining: user.dailyTaps }); } else { res.status(400).json({ message: "Daily Limit Reached" }); } }); // 4. API - Admin Payment Approval app.post('/api/approve-payment', async (req, res) => { const { userId, transactionId } = req.body; // In a real app, you verify 'transactionId' manually or via API const user = await User.findByIdAndUpdate(userId, { isPremium: true, balance: 100 }); res.json({ message: "Premium Unlocked Successfully!" }); }); const PORT = process.env.PORT || 5000; app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
